> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The full token semantics on top of plan 1: device-tagged tokens, encryption at rest with a rotation window, reusable tokens, the enable pair with all three mechanisms, the candidate-set cap, the complete store record set, login context in its final shape with every teardown row and every §5.3 check, accounting annotation, and an in-process mint function that plan 3's minimal broker wraps.

**Architecture:** `internal/token` owns the token format and the `Minter`. `internal/store` grows to the full record set of design §12; `internal/store/crypt` is a decorating `Store` that seals identity and credential fields with AES-256-GCM under a two-key window, so the in-memory store and every future backend hold ciphertext. The plan-1 handlers gain the checks the spec puts on their paths. `internal/session` holds the broker-side operations that touch live context (refresh, revocation) as plain functions over the store, with no HTTP yet.

**Tech Stack:** Go 1.22, standard library (`crypto/aes`, `crypto/cipher`, `crypto/hmac`, `crypto/sha256`), the plan-1 packages. No new dependencies.

**Spec:** `docs/design/2026-08-21-port49-design.md` §8 (all of it), §12 "Store interface", §17 "Minting" and "Requirements" (idempotency only); `docs/design/2026-08-23-policy-engine-spec.md` §5.1.1, §5.1.3, §5.1.4, §5.3, §5.6, §5.8 "Annotating login context", §7, §13 decisions 15 and 17. ## Global Constraints

- Everything in plan 1's Global Constraints still applies (snake_case, closed reason vocabulary, no credentials in the log, no device identity in metrics, Conventional Commits, branch per plan: `feature/token-model`).
- **Token = `tag ‖ secret`**, 6 + 24 base62 characters, no prefix. `tag = base62(HMAC-SHA256(tag_key, device_name))[:6]`. The tag key is the same key material as the encryption-at-rest key set's current key (design §8: "the same place the RFC 8907 shared keys already live"); it is derived, not reused: `tag_key = HMAC-SHA256(current_key, "port49/tag")`.
- **A daemon compares its own tag before any store access.** Mismatch is `device_mismatch` under PAP and ASCII; under CHAP the tag is invisible and the outcome is `token_not_found`.
- **Tokens, receipts, login context, and idempotency records are encrypted at rest.** The store never holds a plaintext secret or a plaintext identity field. Blobs are not encrypted (they contain no identity).
- **Two-key window:** every write uses the current key; every read tries current then previous. Retiring the previous key is a config change, never code.
- **Reusable tokens:** match-without-delete, no receipt, no use counter. Provisional on L1/L7; the context key stays `(nas, port)` until measured.
- **The candidate-set cap refuses the mint, never the redemption**, with a typed error naming the count. Default 8.
- **Accounting annotates and never deletes.** Teardown rows 1–7 of spec §5.6 are the only things that discard context, and row 7 is advisory.
- **Refresh and revocation are broker operations**; the daemon never calls them. Here they are in-process functions with no transport.
- Commit messages end with:
  ```
  Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>
  Claude-Session: https://claude.ai/code/session_01KRDnCbmkhchGxXvV99W4mc
  ```

```go
// Package token defines the device token: a keyed device tag followed by a
// random secret (design §8 "Device binding"), and the Minter that issues them.
package token

import (
	"crypto/hmac"
	"crypto/sha256"
	"io"
	"math/big"
)

const (
	TagLen    = 6
	SecretLen = 24
	Len       = TagLen + SecretLen

	KindLogin  = "login"
	KindEnable = "enable"
)

const alphabet = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"

// TagKey derives the tag key from the current encryption-at-rest key so no new
// secret has to be distributed.
func TagKey(currentEncryptionKey []byte) []byte {
	m := hmac.New(sha256.New, currentEncryptionKey)
	m.Write([]byte("port49/tag"))
	return m.Sum(nil)
}

// Tag is the first six characters of every token minted for deviceName.
func Tag(tagKey []byte, deviceName string) string {
	m := hmac.New(sha256.New, tagKey)
	m.Write([]byte(deviceName))
	return Base62(m.Sum(nil))[:TagLen]
}

// New mints a token: the device's tag followed by SecretLen random characters.
func New(r io.Reader, tagKey []byte, deviceName string) (string, error) {
	secret := make([]byte, SecretLen)
	for i := range secret {
		n, err := randInt(r, len(alphabet))
		if err != nil {
			return "", err
		}
		secret[i] = alphabet[n]
	}
	return Tag(tagKey, deviceName) + string(secret), nil
}

// Parse returns the tag of a presented credential if it has the token shape.
func Parse(presented []byte) (string, bool) {
	if len(presented) != Len {
		return "", false
	}
	for _, c := range presented {
		if !isAlphabet(c) {
			return "", false
		}
	}
	return string(presented[:TagLen]), true
}

func isAlphabet(c byte) bool {
	return (c >= '0' && c <= '9') || (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')
}

// Base62 encodes bytes as a big-endian integer in the alphabet. Not
// length-preserving; used for tags, where only the prefix matters.
func Base62(b []byte) string {
	n := new(big.Int).SetBytes(b)
	if n.Sign() == 0 {
		return "0"
	}
	base := big.NewInt(int64(len(alphabet)))
	var out []byte
	mod := new(big.Int)
	for n.Sign() > 0 {
		n.DivMod(n, base, mod)
		out = append([]byte{alphabet[mod.Int64()]}, out...)
	}
	return string(out)
}

// randInt returns a uniform integer in [0, n) using rejection sampling.
func randInt(r io.Reader, n int) (int, error) {
	limit := 256 - 256%n
	var b [1]byte
	for {
		if _, err := io.ReadFull(r, b[:]); err != nil {
			return 0, err
		}
		if int(b[0]) < limit {
			return int(b[0]) % n, nil
		}
	}
}
```

```go
func (s *Store) live(k store.TokenKey) []store.Token {
	now := s.now()
	out := s.tokens[k][:0]
	for _, t := range s.tokens[k] {
		if t.ExpiresAt.After(now) {
			out = append(out, t)
		}
	}
	if len(out) == 0 {
		delete(s.tokens, k)
	} else {
		s.tokens[k] = out
	}
	return out
}

func (s *Store) PutToken(_ context.Context, k store.TokenKey, t store.Token) (int, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.live(k)
	s.tokens[k] = append(s.tokens[k], t)
	return len(s.tokens[k]), nil
}

func (s *Store) Candidates(_ context.Context, k store.TokenKey) ([]store.Token, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	return append([]store.Token(nil), s.live(k)...), nil
}

func (s *Store) RedeemAndReceipt(_ context.Context, k store.TokenKey, tokenID string, rk store.ReceiptKey, r store.Receipt) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	set := s.live(k)
	for i, t := range set {
		if t.ID == tokenID {
			if !t.Reusable {
				s.tokens[k] = append(set[:i:i], set[i+1:]...)
				if len(s.tokens[k]) == 0 {
					delete(s.tokens, k)
				}
				s.receipts[rk] = r
			}
			return nil
		}
	}
	return store.ErrNotFound
}

func (s *Store) DeleteTokens(_ context.Context, k store.TokenKey) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.tokens, k)
	return nil
}

func (s *Store) GetReceipt(_ context.Context, rk store.ReceiptKey) (store.Receipt, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	r, ok := s.receipts[rk]
	if !ok || !r.ExpiresAt.After(s.now()) {
		delete(s.receipts, rk)
		return store.Receipt{}, store.ErrNotFound
	}
	return r, nil
}

	defer s.mu.Unlock()

	return nil

func (s *Store) GetContext(_ context.Context, k store.ContextKey) (store.Context, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	e, ok := s.contexts[k]
	if !ok || !e.exp.After(s.now()) {

func (s *Store) DeleteContext(_ context.Context, k store.ContextKey) error {
	s.mu.Lock()
	defer s.mu.Unlock()

	return nil

func (s *Store) PutBlob(_ context.Context, hash string, body []byte, ttl time.Duration) error {

	defer s.mu.Unlock()
	exp := s.now().Add(ttl)
	if e, ok := s.blobs[hash]; ok && e.exp.After(exp) {
		exp = e.exp
	}
	s.blobs[hash] = blobEntry{body: append([]byte(nil), body...), exp: exp}
	return nil

	defer s.mu.Unlock()

	defer s.mu.Unlock()

		return store.ErrNotFound
	}

	}
	return nil
}

	if len(m[key]) == 0 {
		delete(m, key)

		delete(m, key)
	}
```

```go
// Package crypt is the encryption-at-rest layer (design §8). It decorates a
// store.Store so that identity and credential fields are sealed before they
// reach any backend, under a current key with one previous key accepted.
package crypt

import (
	"context"
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"encoding/json"
	"errors"
	"io"
	"time"

	"github.com/rhwendt/port49/internal/store"
)

var ErrUnknownKey = errors.New("crypt: sealed under an unknown key")

type Key struct {
	ID    string
	Bytes [32]byte
}

type Keys struct {
	Current  Key
	Previous *Key
}

// Seal: len(id) ‖ id ‖ nonce ‖ AES-256-GCM(plaintext, aad=id).
func (k Keys) Seal(plaintext []byte) ([]byte, error) {
	return seal(k.Current, plaintext)
}

func seal(key Key, plaintext []byte) ([]byte, error) {
	if len(key.ID) == 0 || len(key.ID) > 255 {
		return nil, errors.New("crypt: key id must be 1..255 bytes")
	}
	block, err := aes.NewCipher(key.Bytes[:])
	if err != nil {
		return nil, err
	}
	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}
	nonce := make([]byte, gcm.NonceSize())
	if _, err := io.ReadFull(rand.Reader, nonce); err != nil {
		return nil, err
	}
	out := append([]byte{byte(len(key.ID))}, key.ID...)
	out = append(out, nonce...)
	return gcm.Seal(out, nonce, plaintext, []byte(key.ID)), nil
}

func (k Keys) Open(sealed []byte) ([]byte, error) {
	if len(sealed) < 1 {
		return nil, ErrUnknownKey
	}
	n := int(sealed[0])
	if len(sealed) < 1+n {
		return nil, ErrUnknownKey
	}
	id := string(sealed[1 : 1+n])
	var key *Key
	switch {
	case id == k.Current.ID:
		key = &k.Current
	case k.Previous != nil && id == k.Previous.ID:
		key = k.Previous
	default:
		return nil, ErrUnknownKey
	}
	block, err := aes.NewCipher(key.Bytes[:])
	if err != nil {
		return nil, err
	}
	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}
	rest := sealed[1+n:]
	if len(rest) < gcm.NonceSize() {
		return nil, errors.New("crypt: short ciphertext")
	}
	return gcm.Open(nil, rest[:gcm.NonceSize()], rest[gcm.NonceSize():], []byte(id))
}

	s, err := w.keys.Seal(rec.Response)
	if err != nil {
		return err
	}
	rec.Response, rec.Sealed = nil, s
	return w.inner.PutIdempotency(ctx, principal, key, rec)
}

	}

	return rec, nil
}
```

```go
package token

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"time"

	"github.com/rhwendt/port49/internal/decision"
	"github.com/rhwendt/port49/internal/store"
)

type Subject struct {
	Principal     string
	Username      string
	TokenTTL      time.Duration
	Reusable      bool
	Enable        string
	MaxPrivilege  uint8
	MaxSessionAge time.Duration
	Credential    store.Credential
	AuthTime      time.Time
}

type Request struct {
	Subject        Subject
	Device         string
	Platform       string
	BlobHash       string
	WithEnable     bool
	TTL            time.Duration
	IdempotencyKey string
	BodyHash       string
}

type Result struct {
	Login     string    `json:"login"`
	Enable    string    `json:"enable,omitempty"`
	ExpiresAt time.Time `json:"expires_at"`
	Username  string    `json:"username"`
}

type ErrTooManyOutstanding struct {
	Device, Username string
	Count, Cap       int
}

func (e *ErrTooManyOutstanding) Error() string {
	return fmt.Sprintf("token: %d unspent tokens for %s on %s (cap %d); stop minting and use the ones you have", e.Count, e.Username, e.Device, e.Cap)
}

var ErrIdempotencyConflict = errors.New("token: idempotency key reused with a different body")

type Minter struct {
	store  store.Store
	log    decision.Emitter
	rand   io.Reader
	tagKey []byte
	now    func() time.Time
	cap    int
}

func NewMinter(s store.Store, e decision.Emitter, r io.Reader, tagKey []byte, now func() time.Time, cap int) *Minter {
	return &Minter{store: s, log: e, rand: r, tagKey: tagKey, now: now, cap: cap}
}

func (m *Minter) Mint(ctx context.Context, r Request) (Result, error) {
	if r.IdempotencyKey != "" {
		if rec, err := m.store.GetIdempotency(ctx, r.Subject.Principal, r.IdempotencyKey); err == nil {
			if rec.BodyHash != r.BodyHash {
				return Result{}, ErrIdempotencyConflict
			}
			var res Result
			if err := json.Unmarshal(rec.Response, &res); err == nil {
				return res, nil
			}
		}
	}

	ttl := r.TTL
	if ttl <= 0 || ttl > r.Subject.TokenTTL {
		ttl = r.Subject.TokenTTL
	}
	key := store.TokenKey{NAS: r.Device, Username: r.Subject.Username}
	if cands, err := m.store.Candidates(ctx, key); err != nil {
		return Result{}, err
	} else if len(cands) >= m.cap {
		return Result{}, &ErrTooManyOutstanding{Device: r.Device, Username: r.Subject.Username, Count: len(cands), Cap: m.cap}
	}

	expires := m.now().Add(ttl)
	res := Result{ExpiresAt: expires, Username: r.Subject.Username}
	var hashes []string

	mint := func(kind string) (string, error) {
		secret, err := New(m.rand, m.tagKey, r.Device)
		if err != nil {
			return "", err
		}
		id, err := New(m.rand, m.tagKey, r.Device) // any random string works as a removal handle
		if err != nil {
			return "", err
		}
		_, err = m.store.PutToken(ctx, key, store.Token{
			ID: id[TagLen:], Secret: []byte(secret), Kind: kind, Reusable: r.Subject.Reusable, ExpiresAt: expires,
			Principal: r.Subject.Principal, Username: r.Subject.Username, BlobHash: r.BlobHash,
			MaxSessionAge: r.Subject.MaxSessionAge, Enable: r.Subject.Enable, MaxPrivilege: r.Subject.MaxPrivilege,
			Credential: r.Subject.Credential, AuthTime: r.Subject.AuthTime,
		})
		if err != nil {
			return "", err
		}
		sum := sha256.Sum256([]byte(secret))
		hashes = append(hashes, hex.EncodeToString(sum[:]))
		return secret, nil
	}

	var err error
	if res.Login, err = mint(KindLogin); err != nil {
		return Result{}, err
	}
	if r.WithEnable && r.Subject.Enable == "token" {
		if res.Enable, err = mint(KindEnable); err != nil {
			return Result{}, err
		}
	}

	if r.IdempotencyKey != "" {
		body, _ := json.Marshal(res)
		_ = m.store.PutIdempotency(ctx, r.Subject.Principal, r.IdempotencyKey,
			store.Idempotency{BodyHash: r.BodyHash, Response: body, ExpiresAt: expires})
	}

	m.log.Emit(decision.Record{TS: m.now(), Event: decision.EventMint, Verdict: decision.VerdictPermit, Reason: decision.GrantsMatched,
		Principal: r.Subject.Principal,
		Auth:      &decision.Auth{Method: r.Subject.Credential.Method, CredentialSHA256: ""},
		Device:    &decision.Device{Name: r.Device, Platform: r.Platform},
		Mint:      &decision.Mint{TokenSHA256: hashes, TTLSeconds: int(ttl.Seconds()), Reusable: r.Subject.Reusable, BlobHash: r.BlobHash}})
	return res, nil
}
```

```go
func (h *Handler) Handle(ctx context.Context, peer authen.Peer, req wire.AcctRequest) wire.AcctReply {
	kv, cmd, cmdArgs, raw := author.ParseArgs(req.Args)
	port := string(req.Port)
	user := authen.Canonical(string(req.User))
	var vector []string
	if cmd != "" {
		vector = append([]string{cmd}, cmdArgs...)
	}
	key := store.ContextKey{NAS: peer.NAS, Port: port}
	c, cerr := h.store.GetContext(ctx, key)

	attribution := "unattributed"
	switch {
	case cerr == nil && c.Username == user:
		attribution = "resolved"
	case cerr == nil && cmd == "" && req.Flags&wire.AcctFlagStart != 0:
		attribution = "foreign" // an exec START under another name on a line port49 authenticated
	}

	rec := decision.Record{TS: h.now(), Event: decision.EventAccount, Flags: flagsName(req.Flags),
		Attribution: attribution, TaskID: kv["task_id"],
		Device:  &decision.Device{NAS: peer.NASAddr, Name: peer.NAS, Platform: peer.Platform},
		Session: &decision.Session{Port: port, RemAddr: string(req.RemAddr), Privilege: req.PrivLvl},
		AcctRecord: &decision.AcctRecord{Raw: raw, Vector: vector, PrivLvl: req.PrivLvl,
			StartTime: num(kv["start_time"]), StopTime: num(kv["stop_time"]), ElapsedTime: num(kv["elapsed_time"])}}
	if attribution == "resolved" {
		rec.Principal = c.Principal
		rec.Session.CtxID = c.CtxID
	}

	if err := h.spool.Offer(rec); err != nil {
		h.metrics.Requests.WithLabelValues("acct", "error").Inc()
		return wire.AcctReply{Status: wire.AcctStatusError, ServerMsg: []byte("port49: accounting spool full")}
	}
	h.metrics.Requests.WithLabelValues("acct", "success").Inc()

	// Annotate exec records only; never delete (spec §5.6 row 7, §5.8).
	if cmd == "" && attribution == "resolved" {
		switch {
		case req.Flags&wire.AcctFlagStart != 0:
			if c.TaskID == "" {
				c.TaskID = kv["task_id"]
			}
			c.AccountingState = ""
			_ = h.store.PutContext(ctx, key, c, h.idleTTL)
		case req.Flags&wire.AcctFlagStop != 0:
			if (c.TaskID != "" && c.TaskID == kv["task_id"]) || c.TaskID == "" {
				c.AccountingState = "closing"
				_ = h.store.PutContext(ctx, key, c, h.idleTTL)
			}
		}
	}
```

```go
// Package session holds the broker-side operations on live login context:
// refresh (spec §5.1.3, §5.6) and revocation (spec §5.6 row 2). The daemon
// never calls these. Plan 7's broker puts them behind HTTP.
package session

import (
	"context"
	"time"

	"github.com/rhwendt/port49/internal/store"
)

type Manager struct {
	store    store.Store
	now      func() time.Time
	idleTTL  time.Duration
	indexTTL time.Duration
}

func New(s store.Store, now func() time.Time, idleTTL, indexTTL time.Duration) *Manager {
	return &Manager{store: s, now: now, idleTTL: idleTTL, indexTTL: indexTTL}
}

func (m *Manager) each(ctx context.Context, refs []store.ContextRef, fn func(*store.Context)) ([]store.ContextRef, error) {
	var touched []store.ContextRef
	for _, ref := range refs {
		key := store.ContextKey{NAS: ref.NAS, Port: ref.Port}
		c, err := m.store.GetContext(ctx, key)
		if err != nil {
			continue // index entry outlived its context; the index TTL reclaims it
		}
		fn(&c)
		if err := m.store.PutContext(ctx, key, c, m.idleTTL); err != nil {
			return touched, err
		}
		touched = append(touched, ref)
	}
	return touched, nil
}

func (m *Manager) Refresh(ctx context.Context, principal string, authTime time.Time, newBlobHash, oldSid, newSid string) (int, error) {
	refs, err := m.store.ContextsByPrincipal(ctx, principal)
	if err != nil {
		return 0, err
	}
	touched, err := m.each(ctx, refs, func(c *store.Context) {
		if c.HasCeiling() {
			window := c.SessionExpires.Sub(c.AuthTime)
			c.SessionExpires = authTime.Add(window)
		}
		c.AuthTime = authTime
		c.RefreshedAt = m.now()
		c.RevokedAt = time.Time{}
		if newBlobHash != "" {
			c.BlobHash = newBlobHash
		}
		if newSid != "" {
			c.Credential.Sid = newSid
		}
	})
	if err != nil {
		return 0, err
	}
	if newSid != "" && newSid != oldSid {
		for _, ref := range touched {
			_ = m.store.ReindexSid(ctx, oldSid, newSid, ref, m.indexTTL)
		}
	}
	return len(touched), nil
}

func (m *Manager) RevokeBySid(ctx context.Context, sid string) ([]store.ContextRef, error) {
	refs, err := m.store.ContextsBySid(ctx, sid)
	if err != nil {
		return nil, err
	}
	return m.mark(ctx, refs)
}

func (m *Manager) RevokeByPrincipal(ctx context.Context, principal string) ([]store.ContextRef, error) {
	refs, err := m.store.ContextsByPrincipal(ctx, principal)
	if err != nil {
		return nil, err
	}
	return m.mark(ctx, refs)
}

func (m *Manager) RevokeByCert(ctx context.Context, issuer, serial string) ([]store.ContextRef, error) {
	refs, err := m.store.ContextsByCert(ctx, store.CertKey(issuer, serial))
	if err != nil {
		return nil, err
	}
	return m.mark(ctx, refs)
}

func (m *Manager) mark(ctx context.Context, refs []store.ContextRef) ([]store.ContextRef, error) {
	return m.each(ctx, refs, func(c *store.Context) { c.RevokedAt = m.now() })
}
```
Check the window arithmetic against the first test: seed has `AuthTime = now-1h`, `SessionExpires = now+7h`, window 8h; after refresh with `authTime = now`, `SessionExpires = now+8h`. The window is computed before `AuthTime` is overwritten.
